REPORT ZTEST24_19.


DATA : BEGIN OF ITAB OCCURS 0,
        NUM1 TYPE I,
        NUM2 TYPE I,
        NUM3 TYPE I,
       END OF ITAB.



DO 10 TIMES.    " 10번 실행(반복)
  ITAB-NUM1 = ITAB-NUM1 + 1.      " 1
  clear itab-num2.

  DO 10 TIMES .
    ITAB-NUM2 = ITAB-NUM2 + 1.    " 1
    ITAB-NUM3 = ITAB-NUM1 * ITAB-NUM2.    " 1 * 1

    append itab.

  ENDDO .

ENDDO .

LOOP AT ITAB INTO ITAB. " 행(work area)에 넣어서 1행씩 출력
  WRITE : / ITAB-NUM1, '*' , ITAB-NUM2 , '=' , ITAB-NUM3 .

  IF sy-tabix mod 10 EQ 0.
    WRITE : /.
  ENDIF.
ENDLOOP.
