FORM display_alv_report.
  gd_repid = sy-repid.

* MESSAGE ID SY-MSGID TYPE SY-MSGTY NUMBER SY-MSGNO

*&---------------------------------------------------------------------*
*&      Form  DATA_RETRIEVAL
*&---------------------------------------------------------------------*
*       Retrieve data form SBOOK table and populate itab it_sbook
*----------------------------------------------------------------------*
FORM data_retrieval.

  " sbook 테이블에서 기본 데이터를 불러와서 it_sbook 테이블 구조 변수에 넣기
  SELECT carrid, connid, luggweight, wunit, loccuram, loccurkey
    FROM sbook
    INTO TABLE @it_sbook
   WHERE carrid IN @s_carrid.


  " it_sbook 테이블 구조 변수의 데이터를 it_collect 테이블 구조에 집계
  LOOP AT it_sbook INTO wa_sbook.
    COLLECT wa_sbook INTO it_collect.
  ENDLOOP.


  " scarr 테이블에서 carrid carrname를 읽어와서 it_scarr 테이블 구조 변수에 넣는다
  SELECT carrid carrname
    FROM scarr
    INTO TABLE it_scarr
   WHERE carrid IN s_carrid.

  " spfli 테이블에서 carrid connid cityfrom cityto 를 읽어와서 it_spfli 에 넣는다.
  SELECT carrid connid cityfrom cityto
    FROM spfli
    INTO TABLE it_spfli
   WHERE carrid IN s_carrid.


  LOOP AT it_collect INTO wa_collect.         " it_collect 테이블 구조 변수를 반복

    CLEAR: wa_move, wa_scarr, wa_spfli.       " clear

    MOVE-CORRESPONDING wa_collect TO wa_move. " 한 행을 읽어와서 wa_move 행구조 변수에 넣는다


    SELECT SINGLE carrname FROM scarr INTO wa_move-carrname   " SELECT SINGLE은 딱 하나만 불러와서 바로 넣는다.
      WHERE carrid = wa_move-carrid.
    IF sy-subrc <> 0 .
      wa_move-carrname = '항공사이름정보없음'.
    ENDIF.


    SELECT SINGLE cityfrom, cityto FROM spfli
      INTO ( @wa_move-cityfrom, @wa_move-cityto )
      WHERE carrid = @wa_move-carrid AND connid = @wa_move-connid.
    IF sy-subrc <> 0 .
      wa_move-cityfrom = '비행기정보없음'.
      wa_move-cityto   = '비행기정보없음'.
    ENDIF.

    APPEND wa_move TO it_move.                " it_move 테이블 변수에 wa_move를 넣는다.

  ENDLOOP.

ENDFORM.                    " DATA_RETRIEVAL
