*&---------------------------------------------------------------------*
*&      Form  DISPLAY_ALV_REPORT
*&---------------------------------------------------------------------*
*       Display report using ALV grid

FORM display_alv_report.
  gd_repid = sy-repid.
  CALL FUNCTION 'REUSE_ALV_GRID_DISPLAY'
    EXPORTING
      i_callback_program = gd_repid
      is_layout          = gd_layout
      it_fieldcat        = fieldcatalog[]
      i_save             = 'X'
    TABLES
      t_outtab           = it_join
    EXCEPTIONS
      program_error      = 1
      OTHERS             = 2.
  IF sy-subrc <> 0.
* MESSAGE ID SY-MSGID TYPE SY-MSGTY NUMBER SY-MSGNO
*         WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
  ENDIF.


ENDFORM.                    " DISPLAY_ALV_REPORT


*&---------------------------------------------------------------------*
*&      Form  DATA_RETRIEVAL
*&---------------------------------------------------------------------*
*       Retrieve data form SBOOK table and populate itab it_sbook
*----------------------------------------------------------------------*
FORM data_retrieval.

  DATA: wa_cellcol TYPE slis_specialcol_alv.

  DATA: lv_where_cond TYPE string.    " 동적 WHERE 절을 위한 변수

  IF r1 = 'X'.                        " 전체 예약
    lv_where_cond = ''.
  ELSEIF r2 = 'X'.                    " 취소 되지 않은 예약
    lv_where_cond = 'cancelled <> ''X'''.
  ELSEIF r3 = 'X'.                    " 취소된 예약
    lv_where_cond = 'cancelled = ''X'''.
  ENDIF.

  SELECT
          c~carrid,
          b~connid,
          b~fldate,
          b~bookid,
          b~customid,
          b~loccuram,
          b~loccurkey,
          b~order_date,
          b~cancelled,
          c~carrname,
          cus~name,
          cus~telephone,
          cus~custtype
    FROM scarr AS c
    INNER JOIN sbook AS b
      ON c~carrid = b~carrid
    INNER JOIN scustom AS cus
      ON cus~id = b~customid
    INTO CORRESPONDING FIELDS OF TABLE @it_join
    WHERE b~customid IN @s_cid
      AND b~carrid = @s_carrid
      AND b~connid = @s_connid
      AND b~fldate = @s_fldate
      AND (lv_where_cond).



    SORT it_join BY carrid connid fldate bookid ASCENDING.

    LOOP AT it_join INTO wa_join.

      CLEAR: wa_join-rowcol, wa_join-cellcol.

      wa_join-days = wa_join-fldate - wa_join-order_date. " 날짜 계산


      IF wa_join-cancelled = 'X'.
        wa_join-rowcol = 'C610'.
      ENDIF.

      IF wa_join-days >= 150.
        CLEAR wa_cellcol.

        wa_cellcol-fieldname = 'DAYS'.
        wa_cellcol-color-col = 5.  " 5
        wa_cellcol-color-int = 0.  " 1
        wa_cellcol-color-inv = 0.  " 0
        APPEND wa_cellcol TO wa_join-cellcol.
      ENDIF.

      MODIFY it_join FROM wa_join.

    ENDLOOP.


ENDFORM.
