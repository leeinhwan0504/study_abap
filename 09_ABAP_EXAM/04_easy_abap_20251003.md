REPORT Z01_01.


PARAMETERS: p_val TYPE i.

DATA(gv_grade) = COND string( WHEN p_val = 100 THEN 'A+'
                              WHEN p_val >= 90 THEN 'A'
                              WHEN p_val >= 80 THEN 'B'
                              ELSE 'C'
                              ).


WRITE: '학점은', gv_grade, '입니다'.
