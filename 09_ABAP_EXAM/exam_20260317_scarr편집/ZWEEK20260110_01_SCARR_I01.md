*----------------------------------------------------------------------*
MODULE user_command_0100 INPUT.
  CASE ok_code.
    WHEN 'BACK' OR 'CANC'.
      LEAVE TO SCREEN 0.
    WHEN 'EXIT'.
      LEAVE PROGRAM.
  ENDCASE.
ENDMODULE.
*&---------------------------------------------------------------------*
*&      Module  SAVE_DATA  INPUT
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
MODULE save_data INPUT.
  CASE ok_code.
    WHEN 'SAVE'.
      IF r1 = 'X'.
        PERFORM save_zsc_data.
      ELSEIF r2 = 'X'.
        PERFORM save_modify.
      ENDIF.
  ENDCASE.
ENDMODULE.

MODULE edit_data INPUT.

  CASE ok_code.
    WHEN 'EDIT'.
      IF go_grid->is_ready_for_input( ) = 0.      " 조회모드이면
        CALL METHOD go_grid->set_ready_for_input
          EXPORTING
            i_ready_for_input = 1.                " 편집모드로
      ELSE.                                       " 편집모드이면
        CALL METHOD go_grid->set_ready_for_input
          EXPORTING
            i_ready_for_input = 0.                " 조회모드로
      ENDIF.
  ENDCASE.

ENDMODULE.

MODULE sall_data INPUT.

  CASE ok_code.
    WHEN 'SALL'.

      DATA : lt_index_rows1 TYPE lvc_t_row,
             ls_index_rows  TYPE lvc_s_row,
             lt_row_no      TYPE lvc_t_roid,
             ls_row_no      TYPE lvc_s_roid,
             i              TYPE i.


*    i = 1.

      DESCRIBE TABLE gt_zscarr LINES DATA(line).

      DO line TIMES.
        ls_index_rows-index   = sy-index.
        ls_row_no-row_id      = sy-index.
        ls_row_no-sub_row_id  = 0.

        APPEND : ls_index_rows  TO lt_index_rows1,
                 ls_row_no      TO lt_row_no.

      ENDDO.


      CALL METHOD go_grid->set_selected_rows
       EXPORTING
         it_index_rows = lt_index_rows1
         it_row_no     = lt_row_no.
*         is_keep_other_selections


    WHEN 'DALL'.
      REFRESH : lt_index_rows1,lt_row_no.

      CALL METHOD go_grid->set_selected_rows
        EXPORTING
          it_index_rows = lt_index_rows1
          it_row_no     = lt_row_no.

  ENDCASE.

ENDMODULE.
